export type ID = string;
export type UserRole = "ADMIN" | "MANAGER" | "STAFF";
export interface BaseEntity {
id: ID;
createdAt: Date;
updatedAt: Date;
}
export interface ApiResponse<T> {
success: boolean;
data?: T;
message: string;
timestamp: Date;
}
